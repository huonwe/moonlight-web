# ============================================================================
# mw-click-target — the "ideal game" of plan « attente » (A1).
#
# On a click it draws the latency flag (the bands and the place of
# backend/src/LatencyFlag.cpp) and presents at once, then logs when it got the
# click, when it presented and when the OS says the picture reached the screen,
# on the clock MoonlightWeb stamps with. What a game adds to the click's way
# through the host is then the game's own, not the bench's.
#
# Windows (D3D12, flip model, tearing allowed), macOS (Metal, a CAMetalLayer)
# and Linux (Vulkan on Wayland, wp_presentation), on the same options and log.
# Nothing but cleared or copied rectangles is drawn: no shader.
#
# Built on demand with -DMW_BUILD_TOOLS=ON, never installed nor shipped. It
# also configures on its own: `cmake -S backend/native-host/tools/click-target`.
# ============================================================================

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    cmake_minimum_required(VERSION 3.21)
    if(APPLE)
        project(mw-click-target LANGUAGES CXX OBJCXX)
    else()
        project(mw-click-target LANGUAGES CXX)
    endif()
endif()

if(APPLE)
    add_executable(mw-click-target src/main.cpp src/ClickTargetMac.mm)
    target_compile_features(mw-click-target PRIVATE cxx_std_17)
    target_compile_options(mw-click-target PRIVATE -Wall -Wextra
                                                   $<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>)
    target_link_libraries(mw-click-target PRIVATE "-framework AppKit" "-framework Metal"
                                                  "-framework QuartzCore" "-framework CoreGraphics")
    return()
endif()

if(UNIX)
    # Vulkan on Wayland: xdg-shell for the window, presentation-time for when
    # the picture reached the screen, generated by wayland-scanner.
    find_package(Vulkan QUIET)
    find_package(PkgConfig QUIET)
    if(PkgConfig_FOUND)
        pkg_check_modules(MW_CT_WAYLAND QUIET IMPORTED_TARGET wayland-client)
        pkg_get_variable(MW_CT_PROTOCOLS wayland-protocols pkgdatadir)
    endif()
    find_program(MW_CT_SCANNER wayland-scanner)
    if(NOT Vulkan_FOUND OR NOT MW_CT_WAYLAND_FOUND OR NOT MW_CT_PROTOCOLS OR NOT MW_CT_SCANNER)
        message(STATUS "mw-click-target skipped: needs Vulkan, wayland-client, wayland-protocols "
                       "and wayland-scanner")
        return()
    endif()
    enable_language(C)
    set(gen ${CMAKE_CURRENT_BINARY_DIR}/wayland)
    set(protocol_sources)
    foreach(protocol xdg-shell presentation-time)
        set(xml ${MW_CT_PROTOCOLS}/stable/${protocol}/${protocol}.xml)
        add_custom_command(
            OUTPUT ${gen}/${protocol}-client-protocol.h ${gen}/${protocol}-protocol.c
            COMMAND ${CMAKE_COMMAND} -E make_directory ${gen}
            COMMAND ${MW_CT_SCANNER} client-header ${xml} ${gen}/${protocol}-client-protocol.h
            COMMAND ${MW_CT_SCANNER} private-code ${xml} ${gen}/${protocol}-protocol.c
            DEPENDS ${xml}
            VERBATIM)
        list(APPEND protocol_sources ${gen}/${protocol}-client-protocol.h
             ${gen}/${protocol}-protocol.c)
    endforeach()
    add_executable(mw-click-target src/main.cpp src/ClickTargetLinux.cpp ${protocol_sources})
    target_include_directories(mw-click-target PRIVATE ${gen})
    target_compile_features(mw-click-target PRIVATE cxx_std_17)
    # Vulkan's structures are filled from their sType on: the rest is zero.
    target_compile_options(mw-click-target
                           PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall -Wextra -Wno-missing-field-initializers>)
    target_link_libraries(mw-click-target PRIVATE Vulkan::Vulkan PkgConfig::MW_CT_WAYLAND)
    return()
endif()

if(NOT WIN32)
    return()
endif()

add_executable(mw-click-target src/main.cpp src/ClickTargetWin.cpp)
target_compile_features(mw-click-target PRIVATE cxx_std_17)
target_compile_definitions(mw-click-target PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE
                                                   _CRT_SECURE_NO_WARNINGS)
target_link_libraries(mw-click-target PRIVATE d3d12 dxgi user32 winmm)
if(MSVC)
    target_compile_options(mw-click-target PRIVATE /W4 /utf-8)
endif()
